SQL:
-- Q09: 메뉴 카테고리별 등록 메뉴 수와 평균 가격 집계 (COUNT, AVG)
SELECT category, COUNT(*) AS menu_count, ROUND(AVG(price), 2) AS avg_price
FROM menu
GROUP BY category
ORDER BY avg_price DESC;
