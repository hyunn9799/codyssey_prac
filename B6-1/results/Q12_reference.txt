-- Q12: 전체 메뉴의 평균 가격보다 비싼 메뉴 목록 조회 (서브쿼리)
SELECT menu_id, name, category, price
FROM menu
WHERE price > (SELECT AVG(price) FROM menu)
ORDER BY price DESC;
