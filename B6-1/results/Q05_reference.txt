-- Q05: 주문 목록과 해당 주문을 진행한 고객의 이름 조회 (INNER JOIN)
SELECT o.order_id, c.customer_id, c.name AS customer_name, o.ordered_at, o.status
FROM orders o
INNER JOIN customer c ON o.customer_id = c.customer_id
ORDER BY o.order_id ASC;
